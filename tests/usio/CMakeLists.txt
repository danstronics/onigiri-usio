cmake_minimum_required(VERSION 3.20)
project(OnigiriUSIOHostTests C)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(FIRMWARE ${CMAKE_CURRENT_LIST_DIR}/../../firmware)
add_executable(protocol-test protocol_test.c ${FIRMWARE}/controls.c ${FIRMWARE}/usio_profiles.c ${FIRMWARE}/usio_protocol.c ${FIRMWARE}/selector.c ${FIRMWARE}/card_reader.c)
target_include_directories(protocol-test PRIVATE ${FIRMWARE})
target_compile_options(protocol-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME protocol COMMAND protocol-test)

foreach(FLASH_MB 2 16)
  math(EXPR FLASH_BYTES "${FLASH_MB} * 1024 * 1024")
  add_executable(flash-test-${FLASH_MB} flash_test.c ${FIRMWARE}/usio_flash.c)
  target_compile_definitions(flash-test-${FLASH_MB} PRIVATE ONIGIRI_HOST_TEST PICO_FLASH_SIZE_BYTES=${FLASH_BYTES})
  target_include_directories(flash-test-${FLASH_MB} PRIVATE stubs ${FIRMWARE})
  target_compile_options(flash-test-${FLASH_MB} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME flash-${FLASH_MB} COMMAND flash-test-${FLASH_MB})
endforeach()
set(USIO_TINYUSB_PATH "" CACHE PATH "TinyUSB checkout from Pico SDK 2.2.0, for real USB stack tests")
if(USIO_TINYUSB_PATH)
  add_executable(usb-transport-test usb_transport_test.c ${FIRMWARE}/usio_usb.c
    ${FIRMWARE}/controls.c ${FIRMWARE}/usio_profiles.c ${FIRMWARE}/usio_protocol.c ${FIRMWARE}/card_reader.c
    ${USIO_TINYUSB_PATH}/src/tusb.c ${USIO_TINYUSB_PATH}/src/device/usbd.c
    ${USIO_TINYUSB_PATH}/src/device/usbd_control.c ${USIO_TINYUSB_PATH}/src/common/tusb_fifo.c)
  target_include_directories(usb-transport-test PRIVATE ${FIRMWARE} ${USIO_TINYUSB_PATH}/src)
  target_compile_definitions(usb-transport-test PRIVATE USIO_HOST_TEST_CLOCK USIO_TRACE_BUFFER_ONLY CFG_TUSB_MCU=OPT_MCU_NONE TUP_DCD_ENDPOINT_MAX=16)
  target_compile_options(usb-transport-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME usb-transport COMMAND usb-transport-test)
endif()

add_executable(ui-test ${CMAKE_CURRENT_LIST_DIR}/../ui/ui_preview.c ${FIRMWARE}/ui.c ${FIRMWARE}/usio_profiles.c)
target_include_directories(ui-test PRIVATE ${FIRMWARE})
target_compile_options(ui-test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
# Show the firmware version in rendered screens (used by the manual).
file(STRINGS ${FIRMWARE}/CMakeLists.txt USIO_PROJECT_LINE REGEX "OnigiriUSIO VERSION")
string(REGEX MATCH "[0-9]+[.][0-9]+[.][0-9]+" USIO_VERSION "${USIO_PROJECT_LINE}")
target_compile_definitions(ui-test PRIVATE ONIGIRI_VERSION="${USIO_VERSION}")
add_test(NAME ui-render COMMAND ui-test)
