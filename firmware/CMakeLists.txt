cmake_minimum_required(VERSION 3.20)
set(USIO_SOURCE_DIR ${CMAKE_CURRENT_LIST_DIR})
if(NOT PICO_BOARD)
  set(PICO_BOARD waveshare_rp2040_zero CACHE STRING "RP2040 board")
endif()
include(${USIO_SOURCE_DIR}/../pico_sdk_import.cmake)
project(OnigiriUSIO VERSION 0.3.27 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_subdirectory(${USIO_SOURCE_DIR}/../libs/pico_ssd1306 ${CMAKE_CURRENT_BINARY_DIR}/ssd1306)
set(USIO_SOURCES main.c controls.c selector.c usio_profiles.c usio_protocol.c usio_usb.c usio_flash.c ui.c card_reader.c)
list(TRANSFORM USIO_SOURCES PREPEND "${USIO_SOURCE_DIR}/")
add_executable(Onigiri-USIO ${USIO_SOURCES})
target_include_directories(Onigiri-USIO PRIVATE ${USIO_SOURCE_DIR})
target_compile_options(Onigiri-USIO PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(Onigiri-USIO PRIVATE ONIGIRI_VERSION="${PROJECT_VERSION}")
target_link_libraries(Onigiri-USIO PRIVATE pico_stdlib pico_multicore pico_flash hardware_flash hardware_i2c tinyusb_device tinyusb_board pico_ssd1306)
pico_enable_stdio_usb(Onigiri-USIO 0)
pico_enable_stdio_uart(Onigiri-USIO 0)
pico_add_extra_outputs(Onigiri-USIO)
